# MSIT Website — SEO & Metadata Migration Plan

> **Goal:** Zero SEO regression.  
> **Source Mechanism:** `react-helmet-async` + `scripts/prerender.js` + `vite-plugin-sitemap`  
> **Target Mechanism:** Native Next.js 15+ App Router Metadata API + `app/sitemap.ts` + `app/robots.ts`  
> **Status:** PREPARATION & AUDIT ONLY  

---

## 1. SEO Inventory Comparison

| Feature | Current Vite SPA Implementation | Next.js Target Architecture | Regressive Risk? |
| :--- | :--- | :--- | :--- |
| **Title Tags** | `SEO.jsx` via `react-helmet-async` + regex injection in `prerender.js` | `export const metadata: Metadata` or `generateMetadata()` | **Zero Risk** (Server renders directly in `<head>`) |
| **Meta Descriptions** | `SEO.jsx` + regex injection in `prerender.js` | Native `description` field in Next.js `Metadata` | **Zero Risk** |
| **Canonical URLs** | Injected via `prerender.js`: `<link rel="canonical" href="..." />` + hreflang | `alternates: { canonical: url, languages: { 'en-IN': url, 'x-default': url } }` | **Zero Risk** |
| **Open Graph** | `og:type`, `og:title`, `og:description`, `og:url`, `og:image`, `og:site_name` | `openGraph: { type: 'website', title, description, url, siteName, images }` | **Zero Risk** |
| **Twitter / X Cards** | `twitter:card="summary_large_image"`, `twitter:title`, `twitter:description`, `twitter:image` | `twitter: { card: 'summary_large_image', title, description, images }` | **Zero Risk** |
| **JSON-LD Schemas** | Static JSON scripts in `index.html`, `SEO.jsx`, and `prerender.js` | Inline `<script type="application/ld+json">` rendered in Server Layout and Page | **Zero Risk** |
| **Robots.txt** | Generated by `scripts/prerender.js` pointing to sitemap | Native `app/robots.ts` returning typed `MetadataRoute.Robots` | **Zero Risk** |
| **Sitemap.xml** | `vite-plugin-sitemap` configured in `vite.config.js` | Native `app/sitemap.ts` returning typed `MetadataRoute.Sitemap` | **Zero Risk** |
| **Site Verification** | Google Site Verification tag in `index.html` | `verification: { google: "-a0wyjaTybF3gldEtwwHLwq_ChLau7TLls8Q1KFF7lE" }` in root layout metadata | **Zero Risk** |

## 3. Dynamic Route Metadata (`app/[slug]/page.tsx`)

In the dynamic content engine, metadata is dynamically generated at build time using `generateMetadata`:

```tsx
import type { Metadata } from 'next';
import { pagesData } from '@/lib/data/pagesData';
import { notFound } from 'next/navigation';

type Props = {
  params: Promise<{ slug: string }>;
};

export async function generateMetadata({ params }: Props): Promise<Metadata> {
  const { slug } = await params;
  const page = pagesData[slug];
  if (!page) return {};

  const siteUrl = process.env.NEXT_PUBLIC_SITE_URL || 'https://msit-website.netlify.app';
  const canonicalUrl = `${siteUrl}/${slug}`;
  const description = page.seo_description || page.subtitle || '';

  return {
    title: page.title,
    description: description,
    alternates: {
      canonical: canonicalUrl,
      languages: {
        'en-IN': canonicalUrl,
        'x-default': canonicalUrl,
      },
    },
    openGraph: {
      title: `${page.title} | MSIT`,
      description: description,
      url: canonicalUrl,
      images: page.heroImage ? [{ url: `${siteUrl}${page.heroImage}` }] : undefined,
    },
  };
}
```

## 5. Native Next.js Sitemap & Robots Implementation

### `app/sitemap.ts`
Replaces `vite-plugin-sitemap`:
```ts
import { MetadataRoute } from 'next';
import { pagesData } from '@/lib/data/pagesData';

export default function sitemap(): MetadataRoute.Sitemap {
  const siteUrl = process.env.NEXT_PUBLIC_SITE_URL || 'https://msit-website.netlify.app';
  
  // 1. Static Core Routes
  const staticRoutes: MetadataRoute.Sitemap = [
    { url: `${siteUrl}/`, changeFrequency: 'weekly', priority: 1.0, lastModified: new Date() },
    { url: `${siteUrl}/placements`, changeFrequency: 'weekly', priority: 0.8, lastModified: new Date() },
    { url: `${siteUrl}/facilities`, changeFrequency: 'monthly', priority: 0.8, lastModified: new Date() },
    { url: `${siteUrl}/faculty`, changeFrequency: 'monthly', priority: 0.8, lastModified: new Date() },
    { url: `${siteUrl}/virtual-tour`, changeFrequency: 'monthly', priority: 0.6, lastModified: new Date() },
    { url: `${siteUrl}/academic-calendar`, changeFrequency: 'monthly', priority: 0.6, lastModified: new Date() },
    { url: `${siteUrl}/syllabus`, changeFrequency: 'monthly', priority: 0.6, lastModified: new Date() },
    { url: `${siteUrl}/timetable`, changeFrequency: 'monthly', priority: 0.6, lastModified: new Date() },
    { url: `${siteUrl}/brochure`, changeFrequency: 'monthly', priority: 0.6, lastModified: new Date() },
    { url: `${siteUrl}/team`, changeFrequency: 'monthly', priority: 0.6, lastModified: new Date() },
    { url: `${siteUrl}/privacy`, changeFrequency: 'yearly', priority: 0.3, lastModified: new Date() },
    { url: `${siteUrl}/terms`, changeFrequency: 'yearly', priority: 0.3, lastModified: new Date() },
  ];

  // 2. Dynamic Slugs from pagesData
  const dynamicRoutes: MetadataRoute.Sitemap = Object.keys(pagesData).map((slug) => {
    const isDept = ['cse', 'it', 'ece', 'eee', 'applied-sciences'].includes(slug);
    return {
      url: `${siteUrl}/${slug}`,
      changeFrequency: isDept ? 'monthly' : 'monthly',
      priority: isDept ? 0.8 : 0.6,
      lastModified: new Date(),
    };
  });

  return [...staticRoutes, ...dynamicRoutes];
}
```

### `app/robots.ts`
Replaces robots generation in `scripts/prerender.js`:
```ts
import { MetadataRoute } from 'next';

export default function robots(): MetadataRoute.Robots {
  const siteUrl = process.env.NEXT_PUBLIC_SITE_URL || 'https://msit-website.netlify.app';

  return {
    rules: {
      userAgent: '*',
      allow: '/',
      disallow: ['/admin', '/admin/', '/.netlify', '/500', '/search', '/api/'],
    },
    sitemap: `${siteUrl}/sitemap.xml`,
  };
}
```
